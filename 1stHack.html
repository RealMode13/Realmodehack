        // Get Premium Member Count

        function getPremiumMemberCount(number) {

            let baseCount = Math.floor(Math.random() * 80) + 70;

            if (number === state.currentMemberChoice) baseCount += 100;

        function updateRealTimeClock() {

            const now = new Date();

            const hours = now.getHours().toString().padStart(2, '0');

            const minutes = now.getMinutes().toString().padStart(2, '0');

            const seconds = now.getSeconds().toString().padStart(2, '0');

           

            document.getElementById('realTimeClock').textContent =

                `${hours}:${minutes}:${seconds}`;

           

            // Check if it's time to update period (:00 seconds)

            if (seconds === '00') {

                updatePeriodNumber();

            }

        }

        // Calculate Next Period Time

        function calculateNextPeriodTime() {

            const now = new Date();

            const currentSeconds = now.getSeconds();

            const secondsToNextMinute = 60 - currentSeconds;

           

            state.nextPeriodTime = new Date(now.getTime() + secondsToNextMinute * 1000);

        }

        // Calculate Next Prediction Time

        function calculateNextPredictionTime() {

            const now = new Date();

            const nextPrediction = new Date(now.getTime() + 60000); // 1 minute from now

            state.nextPredictionTime = nextPrediction;

        }

        // Update Next Period Timer

        function updateNextPeriodTimer() {

            if (!state.nextPeriodTime) {

                calculateNextPeriodTime();

                return;

            }

           

            const now = new Date();

            const diff = Math.floor((state.nextPeriodTime - now) / 1000);

           

            if (diff <= 0) {

                calculateNextPeriodTime();

                updateNextPeriodTimer();

                return;

            }

           

            const minutes = Math.floor(diff / 60);

            const seconds = diff % 60;

           

            document.getElementById('nextPeriodTimer').textContent =

                `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;

           

            document.getElementById('nextUpdateSeconds').textContent = seconds;

           

            // Update next period number display

            updateNextPeriodNumber();

        }

        // Update Prediction Timer

        function updatePredictionTimer() {

            if (!state.nextPredictionTime) {

                calculateNextPredictionTime();

                return;

            }

           

            const now = new Date();

            const diff = Math.floor((state.nextPredictionTime - now) / 1000);

           

            if (diff <= 0) {

                autoUpdatePrediction();

                calculateNextPredictionTime();

                return;

            }

           

            document.getElementById('nextPredictionTimer').textContent = `${diff}s`;

        }

        // Update Period Number (at :00 seconds)

        function updatePeriodNumber() {

            // Extract the sequence number part (last 5 digits)

            const sequencePart = state.currentPeriod.substring(12);

            const sequenceNum = parseInt(sequencePart);

           

            // Increment sequence number

            const newSequenceNum = sequenceNum + 1;

           

            // Update the period number with new sequence

            state.currentPeriod = state.currentPeriod.substring(0, 12) +

                                 newSequenceNum.toString().padStart(5, '0');

           

            document.getElementById('currentPeriod').textContent = state.currentPeriod;

           

            // Update last sync time

            state.lastSyncTime = new Date();

            updateLastSyncTime();

           

            // Update next period number

            updateNextPeriodNumber();

           

            // Show notification

            showNotification(`Period updated to: ${state.currentPeriod}`);

        }

        // Update Last Sync Time Display

        function updateLastSyncTime() {

            const now = new Date();

            const diff = Math.floor((now - state.lastSyncTime) / 1000);

           

            let timeText;

            if (diff < 60) {

                timeText = 'Just now';

            } else if (diff < 120) {

                timeText = '1 minute ago';

            } else if (diff < 3600) {

                timeText = `${Math.floor(diff / 60)} minutes ago`;

            } else {

                timeText = `${Math.floor(diff / 3600)} hours ago`;

            }

           

            document.getElementById('lastSyncTime').textContent = `Last sync: ${timeText}`;

        }
